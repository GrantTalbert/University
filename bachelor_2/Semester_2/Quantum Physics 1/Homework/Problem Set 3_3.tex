\section{Representations and Basis States}
\begin{enumerate}
	\item Even though our space is infinite-dimensional, we can isolate $v_1,v_2$ to consider a 2-dimensional subspace. The most general possible choice in this subspace, as illustrated by equation (1.1), is
\[
	\ket{1}=\cos \theta e^{i\phi_1} v_1+\sin \theta e^{i\phi_2} v_2,
\]
\[
	\ket{2}=\sin \theta e^{i\phi_3} v_1-\cos \theta e^{i\left( \phi_3+\phi_1-\phi_2 \right) } v_2
.\]
\item Note that due to orthonormality, for $v\in\{\ket{3},\ket{4} \} $,
		\[
			\braket{v}{1} =\frac{1}{2} \sum_{i} v_i=0
		.\]
		Similarly,
		\[
			\braket{v}{2} =\frac{1}{2}( v_1+v_2-v_3-v_4)=0
		.\]
		Therefore, we have
		\[
			v_1+v_2+v_3+v_4=v_1+v_2-v_3-v_4\implies v_3=-v_4
		.\]
		We then have $v_1+v_2=0$ and thus $v_1=-v_2$. We then have that $\ket{3} , \ket{4} $ must satisfy
		\[
			\begin{pmatrix}
			v_1 \\
			-v_1 \\
			v_3 \\
			-v_3
			\end{pmatrix}
		.\]
		These vectors are determined by two distinct components $v_1,v_3$, and thus we can plug in equation 1.1 to arrive at
		\[
			\ket{3} =\frac{1}{\sqrt{2} } \begin{pmatrix}
			\cos \theta e^{i\phi_1}  \\
			-\cos \theta e^{i\phi _1}  \\
			\sin \theta e^{i\phi _2}  \\
			-\sin \theta e^{i\phi _2} 
			\end{pmatrix},\qquad \ket{4} =\frac{1}{\sqrt{2} } \begin{pmatrix}
			\sin \theta e^{i\phi _3}  \\
			-\sin \theta e^{i\phi _3}  \\
			-\cos \theta e^{i\left( \phi _3-\phi _1+\phi _2 \right) }  \\
			\cos \theta e^{i\left( \phi _3-\phi _1+\phi _2 \right) } 
			\end{pmatrix}
		.\]
		The $1/\sqrt{2} $ is the normalizing factor, which I guess I will show the calculation for.
		\[
			\begin{pmatrix}
			\cos \theta e^{-i\phi _1}  & -\cos \theta e^{-i\phi _1}  & \sin \theta e^{-i\phi _2}  & -\sin \theta e^{-i\phi _2} 
			\end{pmatrix}\begin{pmatrix}
			\cos \theta e^{i\phi _1}  \\
			-\cos \theta e^{i\phi _1}  \\
			\sin \theta e^{i\phi _2}  \\
			-\sin \theta e^{i\phi _2} 
			\end{pmatrix}=\cos^2\theta +\cos ^2\theta +\sin ^2\theta +\sin ^2\theta
		.\]
		This is the norm squared, so the norm is $\sqrt{2} $. I'll calculate the next one as well:
		\[
			\begin{pmatrix}
			\sin \theta e^{-i\phi _3}  & -\sin \theta e^{-i\phi _3}  & -\cos \theta e^{-i\left( \phi _3-\phi _1+\phi _2 \right) }  & \cos \theta e^{-i\left( \phi _3-\phi _1+\phi _2 \right) } 
			\end{pmatrix}\begin{pmatrix}
			\sin \theta e^{i\phi _3}  \\
			-\sin \theta e^{i\phi _3}  \\
			-\cos \theta e^{i\left( \phi _3-\phi _1+\phi _2 \right) }  \\
			\cos \theta e^{i\left( \phi _3-\phi _1+\phi _2 \right) } 
			\end{pmatrix}
		\]
		\[
			=\sin ^2\theta +\sin ^2\theta +\cos ^2\theta +\cos ^2\theta =2
		.\]
		I will also show that they are orthogonal to each other:
\[
	\begin{pmatrix}
			\cos \theta e^{-i\phi _1}  & -\cos \theta e^{-i\phi _1}  & \sin \theta e^{-i\phi _2}  & -\sin \theta e^{-i\phi _2} 
			\end{pmatrix}\begin{pmatrix}
			\sin \theta e^{i\phi _3}  \\
			-\sin \theta e^{i\phi _3}  \\
			-\cos \theta e^{i\left( \phi _3-\phi _1+\phi _2 \right) }  \\
			\cos \theta e^{i\left( \phi _3-\phi _1+\phi _2 \right) } 
			\end{pmatrix}
\]
\[
	=2\cos \theta \sin \theta e^{i(\phi _3-\phi _1)} +2 \cos \theta \sin \theta e^{i(\phi _3-\phi _1)} =2\braket{+}{-} =0
.\]
These kets are also orthogonal to $\ket{1} ,\ket{2} $ because they satisfy the property I listed earlier, and since this is a set of 4 linearly independent vectors in a 4 dimensional vector space, they are necessarily a basis.
\end{enumerate}
\section{Operators}
\begin{enumerate}
	\item Note that if $A$ is Hermitian, then $\bra{p} A\ket{\Omega } =\bra{p} A^{\dagger} \ket{\Omega } $. By contrapositive, if $\bra{p} A\ket{\Omega } \neq \bra{p} A^{\dagger} \ket{\Omega } $, then $A$ is not Hermitian. Note that
		\[
			\bra{p} A\ket{\Omega } =\braket{p}{0} =0
		.\]
		We have
		\[
			\bra{p} A^{\dagger} \ket{\Omega } =\left( \bra{\Omega } A\ket{p}  \right) ^{\dagger} =\left( \braket{\Omega }{\Omega }  \right) ^{\dagger} 
		.\]
		Since $\Omega $ is a state, it must satisfy $\braket{\Omega }{\Omega } =1$, and $1^{\dagger} =1^*=1$. Therefore, $\bra{p} A\ket{\Omega } \neq \bra{p} A^{\dagger} \ket{\Omega } $, and $A$ is not Hermitian.
	\item Note that $h(x)g(x)=0$ (except possibly on $\left\{ 1/3,2/3 \right\} $ depending on how the drawing is interpreted, but this set has lebesgue measure 0 so the following conclusion still holds). It follows that
		\[
			\int_{0}^{1} h(y)g(y)g(x) \,\mathrm{d} y=0
		.\]
		Therefore,
		\[
			(Ah)(x)=\int_{0}^{1} h(y)^2h(x) \,\mathrm{d} y+\int_{0}^{1} h(y)v(y)v(x) \,\mathrm{d} x
		.\]
		Note that $v(x)=\frac{1}{3} h(x)$. It follows that
		\[
			(Ah)(x)=\int_{0}^{1} h(y)^2h(x) \,\mathrm{d} y+\int_{0}^{1} \frac{1}{9} h(y)^2 h(x)\,\mathrm{d} y=\frac{10}{9} h(x)\int_{0}^{1} h(y)^2 \,\mathrm{d} y
		.\]
		Also note that since $0^2=0$ and $1^2=1$, we have that $h(x)^2=h(x)$, so
		\[
			(Ah)(x)=\frac{10}{9} h(x)\int_{0}^{1} h(y) \,\mathrm{d} y=\frac{20}{27} h(x) 
		.\]

		Computing $(Ag)(x)$ is more simple, since the first and last integrals vanish. We have
		\[
			(Ag)(x)=g(x)\int_{0}^{1} g(y)^2 \,\mathrm{d} y=g(x)\int_{1/3}^{2/3}  \,\mathrm{d} x=\frac{1}{3} g(x)
		.\]
	\item By part 2, $20/27$ is an eigenvalue with $h(x)$ as an eigenfunction, and $1/3$ is an eigenvalue with $g(x)$ as an eigenfunction, and any scalar multiple of these functions is also an eigenfunction. Note that the image of $A$ must be a subset of the span of $\left\{ h,g \right\} $ since $v=1/3h$ is in this span, and thus these are clearly the only possible linearly independent eigenspaces other than the 0 eigenspace.
\end{enumerate}

\section{Born's Rule and Expectation Values}
\begin{enumerate}
	\item Recall:
		\[
			S_{\vec{n} } =\begin{pmatrix}
			\cos \theta  & \sin \theta  \\
			\sin \theta  & -\cos \theta 
			\end{pmatrix},
		\]
		and the eigenvectors associated with eigenvales $\hbar /2,-\hbar /2$ are
		\[
			\ket{\psi _1} =\cos \frac{\theta }{2} \ket{+} +\sin \frac{\theta }{2} \ket{-} ,\qquad \ket{\psi _2} =\sin \frac{\theta }{2} \ket{+} -\cos \frac{\theta }{2} \ket{-} ,
		\]
		respectively. We have
		\[
			\mathcal{P}_{\hbar /2} =\left| \braket{+}{\psi_1}  \right|^2=\left| \cos \frac{\theta }{2} \braket{+}{+} +\sin \frac{\theta }{2} \braket{+}{-}  \right| ^2=\left| \cos \frac{\theta }{2}  \right| ^2
		.\]
		Similarly, we have
		\[
			\mathcal{P}_{-\hbar /2} =\left| \braket{+}{\psi_2}  \right| ^2=\left| \sin \frac{\theta }{2} \braket{+}{+} -\cos \frac{\theta }{2} \braket{+}{-}  \right| ^2 = \left| \sin \frac{\theta }{2}  \right|^2
		.\]
		The former is the probability of measuring $\hbar /2$, since $\ket{\psi _1} $ is the corresponding eigenvector, and the latter is the probability of finding $-\hbar /2$ for the same reason. As a quick sanity check,
		\[
			\left| \sin \frac{\theta }{2}  \right| ^2+\left| \cos \frac{\theta }{2}  \right| ^2=\sin ^2\frac{\theta }{2} +\cos ^2\frac{\theta }{2} =1
		.\]
	\item 
		\begin{enumerate}
			\item
				\begin{align*}
					\bra{+} S_{\vec{n} } \ket{+} &=\frac{\hbar }{2} \begin{pmatrix}
					1 & 0
					\end{pmatrix} \begin{pmatrix}
					\cos \theta  & \sin \theta  \\
					\sin \theta  & -\cos \theta 
					\end{pmatrix}\begin{pmatrix}
					1 \\
					0
					\end{pmatrix}\\
								     &=\frac{\hbar }{2} \begin{pmatrix}
								     1 & 0
								     \end{pmatrix}\begin{pmatrix}
								     \cos \theta  \\
								     \sin \theta 
								     \end{pmatrix}\\
								     &=\frac{\hbar }{2} \cos \theta 
				.\end{align*}
			\item 
				\[
					\left\langle S_{\vec{n} }  \right\rangle =\sum_{i} \lambda_i \mathcal{P}_i=\frac{\hbar }{2} \cos ^2\frac{\theta }{2}-\frac{\hbar }{2} \sin ^2\frac{\theta }{2} =\frac{\hbar }{2} \left( \cos^2\frac{\theta }{2} -\sin ^2\frac{\theta }{2}  \right) =\frac{\hbar }{2} \cos \theta 
				.\]
		\end{enumerate}
	\item
		\begin{enumerate}
			\item Note that
				\[
					\mathcal{P}_{\hbar /2} =\left| \braket{\psi _1}{\psi _1}  \right| ^2 \equiv 1
				.\]
				It follows that $\mathcal{P}_{-\hbar /2} =0$ by classical probability theory. Therefore, $\ket{\psi _1} $ and $\ket{\psi _2} $ are orthogonal states.
			\item
				\begin{enumerate}
					\item
						\begin{align*}
							\bra{\psi _1} S_{\vec{n} } \ket{\psi _1} &=\frac{\hbar }{2}\begin{pmatrix}
							\cos \frac{\theta }{2}  & \sin \frac{\theta }{2} 
							\end{pmatrix}\begin{pmatrix}
							\cos \theta  & \sin \theta  \\
							\sin \theta  & -\cos \theta 
							\end{pmatrix}\begin{pmatrix}
							\cos \frac{\theta }{2}  \\
							\sin \frac{\theta }{2} 
							\end{pmatrix}\\
												 &=\frac{\hbar }{2}\begin{pmatrix}
												 \cos \frac{\theta }{2}  & \sin \frac{\theta }{2} 
												 \end{pmatrix}\begin{pmatrix}
												 \cos \theta \cos \frac{\theta }{2} +\sin \theta \sin \frac{\theta }{2}  \\
												 \cos \frac{\theta }{2} \sin \theta -\sin \frac{\theta }{2} \cos \theta 
												 \end{pmatrix}\\
												 &=\frac{\hbar }{2} \left(\cos \theta \cos ^2\frac{\theta }{2} +\sin \theta \cos \frac{\theta }{2} \sin \frac{\theta }{2} +\cos \frac{\theta }{2} \sin \frac{\theta }{2} \sin \theta -\sin ^2\frac{\theta }{2} \cos \theta\right) \\
												 &=\frac{\hbar }{2} \left(\cos \theta \left( \cos ^2\frac{\theta }{2} -\sin ^2\frac{\theta }{2}  \right) +2 \sin \theta \cos \frac{\theta }{2} \sin \frac{\theta }{2}\right) \\
												 &=\frac{\hbar }{2} \left(\cos ^2\theta +\sin ^2\theta\right) \\
												 &=\frac{\hbar }{2} 
						.\end{align*}
					\item
						\[
							\left\langle S_{\vec{n} }  \right\rangle =\sum_{i} \lambda_i \mathcal{P}_{i} =\frac{\hbar }{2} 1-\frac{\hbar }{2} 0=\frac{\hbar }{2} 
						.\]
				\end{enumerate}
		\end{enumerate}
\end{enumerate}
