	\[
		\left( \frac{\hbar }{2} \bra{-}_1  \otimes \frac{-i\hbar }{2} \bra{-}_2   \right) \left( \frac{\hbar }{2} \ket{-}_1 \otimes \frac{i\hbar }{2} \ket{-}_2  \right) =\frac{\hbar ^2}{4} \braket{-}{-}_1 \cdot \frac{\hbar ^2}{4} \braket{-}{-}_2 =\frac{\hbar ^4}{16}

The projector $P_a \otimes \mathbbm{1}$ maps a two particle state $\ket{\psi }=\ket{\psi}_1 \otimes \ket{\psi }_2 $ to the state $(P_a \otimes \mathbbm{1})\ket{\psi } =\ket{a}_1 \otimes \ket{\psi }_2$. The probability of measuring the state $\ket{a}_1 \otimes \ket{\psi }_2$ in some system $\ket{\phi } $, that is, measuring $\ket{a} $ in the first particle and the same state $\ket{\psi }_2$ in the second, is given by the Born rule
\[
	\left| \bra{\phi  }(\ket{a}_1 \otimes \ket{\psi }_2 \right|^2
.\]
This can be reframed as
\[
	\left| \bra{\phi } (P_a \otimes \mathbbm{1})\ket{\psi }  \right| ^2
,\]
since
\[\begin{tikzcd}[row sep = 2.5em, column sep = 2.5em]
		\ket{\psi }_1 \otimes \ket{\psi }_2 \arrow[r, "P_a \otimes \mathbbm{1}"]&\ket{a}_1 \otimes \ket{\psi }_2.
\end{tikzcd}\]
