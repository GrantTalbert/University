\begin{definition}[Ring Homomorphism]\label{dfn:1.11}
	Let $(R,+_R,\bullet_R)$ and $(S,+_S,\bullet _S)$ be rings. A function
	\[
		\phi : R \to S
\]
	is called a \emph{ring homomorphism} if and only if
	\[
		\phi (a+_Rb)=\phi (a)+_S\phi (b),\qquad \phi (a\bullet _Rb)=\phi (a)\bullet _S\phi (b)
	.\]
\end{definition}
\begin{definition}[Isomorphism]\label{dfn:1.12}
	A ring \emph{isomorphism} is a ring homomorphism that is invertible/a bijection. If there exists an isomorphism $R\to S$, then we say $R$ and $S$ are isomorphic, and write $R\cong S$.
\end{definition}
\begin{theorem}\label{thm:1.2}
	Let $E$ be a field with a subfield $F\subseteq E$ and let $\alpha \in E$. Then the evaluation function
	\[
		\phi_\alpha : F[x] \to E
	\]
	is a homomorphism.
\end{theorem}
\begin{proof}
	Recall that $\phi_\alpha $ is the unique extension of the inclusion map $F\hookrightarrow E$ to $F[x]$ that maps $x\mapsto  \alpha $. The proof isn't worth writing down.
\end{proof}
