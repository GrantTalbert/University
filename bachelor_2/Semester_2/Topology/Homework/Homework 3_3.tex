\section*{Problem 2}
Note that if $[(z,w)]\in\mathbb{C} \mathbb{P} ^1$ with $z\neq 0$, then $[(z,w)]=[(1,w/z)]$.  Thus, every such point with $z\neq 0$ can be represented of this form. If $z=0$, then we instead have $[(0,w)]=[(0,w/w)]=[(0,1)]$ since $(z,w)\neq (0,0)$. Any two points $[(1,w/z)],[(1,a/b)]$ are now distinct in $\mathbb{C} \mathbb{P} ^1$ if $w/z\neq a/b$, since multiplying through by a scaling factor would also shift the value of the $1$. We can thus view $\mathbb{C} \mathbb{P} ^1$ as $\mathbb{C} \cup \left\{ [(0,1)] \right\} $.

Let $\phi : \mathbb{C} \mathbb{P}^1 \to S^2$ be the map $\phi (0,1)=(1,0,0)$, and
\[
	\phi (1,z)=\left( \frac{2 \operatorname{Re } z}{\left| z \right|^2+1} ,\frac{2 \operatorname{Im } z}{\left| z^2 \right| +1} ,\frac{\left| z \right| ^2-1}{\left| z^2 \right| +1}  \right) 
.\]
This map is well-defined by the logic I explained previously, identifying each element of $\mathbb{C} \mathbb{P} ^1 $ with an element of the form $[(1,z)]$ or $[(0,1)]$. Let $\psi : S^2 \to \mathbb{C} \mathbb{P} ^1$ be the map
\[
	\psi (x,y,z)=\left[ \left( 1,\frac{x+iy}{1-z}  \right)  \right] 
,\]
and $\psi (0,0,1)=[(0,1)]$. These maps are both obviously well-defined, and a lot of explicit computation shows they are in fact inverses. Therefore, they are bijections.

Since $\phi $ is a composition of continuous functions (rationals, Re, Im, etc.), it is continuous on elements other than $(0,0,1)$.

$\psi $ is seen to be continuous at points other than $(0,0,1)$ for the same reasons. As a point approaches $(0,0,1)$ on $S^2$, we have $x,y$ approaching $0$, and thus $\psi (x,y,z)$ approaches $[(0,1)]$. Therefore, $\psi $ is continuous.

\begin{lemma}\label{lma:1}
	Let $Y\subseteq X$ be a dense subset of some topological space $X$. Then for all nonempty open sets $U\subseteq X$, there exists $y\in U \cap Y$. That is, every nonempty open set contains some element of $Y$.
\end{lemma}
\begin{proof}
	Since $Y$ is dense, by definition $\overline{Y} =X$. Since $\overline{Y} =X$ is the smallest closed set containing $Y$, it follows that no proper closed subset $U\subsetneq X$ contains $Y$. Equivalently, if $U\subsetneq X$ is closed, then $Y\setminus U\neq \varnothing $.

	If $Y \cap U=\varnothing $, then $U^{c} \subseteq Y$, and since $U\neq X$, we have $U^{c} =U^{c} \cap Y\neq \varnothing $, and we are done. If $U$ and $Y$ are \emph{not} disjoint, then $Y\setminus U\neq \varnothing $ since $U$ doesn't contain $Y$. By definition, this implies $Y \cap U^{c} \neq \varnothing $. Since $U^{c} $ is open, there exists $y\in Y \cap U^{c} $. Since every nonempty open set can be represented as $U^{c} $ for $U\subsetneq X$ closed, we have that every nonempty open set $U\subseteq X$ has some $y\in Y \cap U$.
\end{proof}

\section*{Problem 7}
Let $x\in X$. We have $\overline{d} (x,x)=\min \left\{ 0,1 \right\} =0$. Moreover, if $\overline{d} (x,y)=0$ for some $x,y\in X$, then $\min \left\{ d(x,y),1 \right\} =0$, and thus $d(x,y)=0$, and thus $x=y$. Therefore, $\overline{d} (x,y)=0$ if and only if $x=y$.

Let $x,y\in X$. Since $d$ is a metric, $d(x,y)=d(y,x)$, and thus
\[
	\overline{d} (x,y)=\min \left\{ d(x,y),1 \right\} =\min \left\{ d(y,x),1 \right\} =\overline{d} (y,x)
.\]
Therefore, $\overline{d} (x,y)=\overline{d} (y,x)$.

Finally, let $x,y,z\in X$. We have
\[
	\overline{d} (x,y)+\overline{d} (y,z)=\min \left\{ d(x,y),1 \right\} +\min \left\{ d(y,z),1 \right\} 
.\]
We will brute force this with four cases.
\begin{itemize}
	\item Suppose $d(x,y),d(y,z)<1$. Then since $d(x,y)+d(y,z)\geq d(x,z)$, we have $d(x,z)<1$, and thus
		\[
			\overline{d} (x,z)=d(x,z),\qquad \overline{d} (x,y)=d(x,y),\qquad \overline{d} (y,z)=d(y,z)
		.\]
		Therefore, $\overline{d} (x,z)\leq \overline{d} (x,y)+\overline{d} (y,z)$.
	\item Suppose $d(x,y)\geq 1>d(y,z)$. Then
		\[
			d(x,y)+d(y,z)\geq 1+d(y,z)>1
		.\]
		Since $\overline{d} (x,z)\leq 1$ for all $x,z\in X$, we have that
		\[
			\overline{d} (x,y)+\overline{d} (y,z)\geq \overline{d} (x,z)
		.\]
	\item Suppose $d(y,z)\geq 1>d(x,y)$. By the same logic as previously, the result follows.
	\item Suppose $d(x,y),d(y,z)\geq 1$. Then $\overline{d} (x,y)+\overline{d} (y,z)=2$. If $d(x,z)<1$, then $\overline{d} (x,z)=d(x,z)<1<2$. If not, then $\overline{d} (x,z)=1<2$. Therefore,
		\[
			\overline{d} (x,z)\leq \overline{d} (x,z)+\overline{d} (y,z)
		.\]
\end{itemize}
Therefore, $\overline{d} $ is a metric on $X$.

Note that the basis $\mathcal{B}_{\overline{d} } \subseteq \mathcal{B}_d$, so the topology induced by $\overline{d} $ is a subset of the topology induced by $d$. It thus suffices to show $\mathcal{B}_d \subseteq \mathcal{T}_{\overline{d} } $. Let $B_r^d(x)$ be an open ball in the metric $d$. For any $z\in B_\ell^d(y)$, we have
\[

.\]
Therefore, $z\in B_r^d(x)$, and $B_\ell ^d(y)\subseteq B_r^d(x)$.

Using this same notation, note that $\overline{d}(x,y)\leq d(x,y)$ for all $x,y\in X$, since the only possibilities are that $d(x,y)=\overline{d} (x,y)$, or $\overline{d}(x,y)=1$ when $d(x,y)\geq 1$. We thus have

Therefore, $\mathcal{T}_d \subseteq  \mathcal{T}_{\overline{d} } $, and they must induce the same topology.
