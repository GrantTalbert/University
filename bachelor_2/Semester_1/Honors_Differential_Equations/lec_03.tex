This makes it way easier:
\[
	\int y \,\mathrm{d} y=\int \frac{t}{1-t^2} \,\mathrm{d} t

\[
	\frac{1}{2}y^2=-\int \frac{1}{u} \,\mathrm{d} u
.\]
\[
	\frac{y^2}{2}=\ln \left| 1-t^2 \right| +C
.\]
Solving from here is trivial.

An important problem is the \textbf{mixing} \textbf{problem}. Some container full of some solution, adding more stuff from the top while also draining out some stuff from the bottom. Rates of both can be diff. Stirring all the time to make sure things are mixed.
\begin{example}
	Tank have 50gal of water. At $t=0$, the tank contains pure water. Salty brine is poured into the tank at a rate of 3gal/min with a concentration of 1lb/gal of salt. Brine is drained from the tank at a rate of 3gal/min. How much salt is in the tank after 5 minutes?
\end{example}
\begin{solution}
	Independent variable is time and dependent variable is amount of salt. Alternatively, salt concentration can be used (exercise to reader?)

	Let $S(t)$ be the amount, in lbs, of salt at some time $t$. Since we are adding 1 lb per gallon and 3 gallons per minute, we add 3 lbs per minute. This is a constant positive rate of change of $+3$. We are then removing the amount of water being removed (3 gallons) times the concentration of salt in the tank, which is the amount of salt divided by 50 gallons.
	\[
		\frac{\mathrm{d}S}{\mathrm{d}t} =3-3\left( \frac{S(t)}{50} \right) 
	.\]
	The first term is adding salt, and the second term is removing salt. Simplifying, we get
	\[
		\frac{\mathrm{d}S}{\mathrm{d}t} =3\left( 1-\frac{S(t)}{50} \right) =\frac{3(50-S(t))}{50}
	.\]
	This is separable.
	\[
		\int \frac{1}{50-S(t)} \,\mathrm{d} S=\int \frac{3}{50} \,\mathrm{d} t
	.\]
	\[
		-\ln \left| 50-S(t) \right| =\frac{3t}{50}+C
	.\]
	\[
		S(t)=50-e^{-C}e^{-\frac{3t}{50}}
	.\]
	\[
		S(0)=0=50-e^{-C}
	.\]
	Therefore, $e^{-C}=50$, and $S(t)=50-50\exp \left( -\frac{3t}{50} \right) $.
	Plug in the time asked for.
	\[
		S(5)=50\left( 1-e^{-\frac{15}{50}} \right) \text{lbs}
	.\]
\end{solution}
\section{Euler's Method and Taylor's Theorem}

Recall Taylor Polynomials. We know that an approximation for $f(t)$ about some $a\in\mathbb{R}$ is given by the Taylor Polynomial about $a$,
\[
	f(t)=\sum_{n=0}^{\infty} \frac{f^{(n)}(a)}{n!}(t-a)^n
.\]
A taylor polynomial of order $n$ will be called $T_n(t)$.
\begin{prev}
	Euler's Method:

	Take an interval split into a grid of points $t_j$, separated by value $\Delta t$. Suppose $y(t_j)$ is known, at least approximately. Define $y(t_{j+1})=y(t_j+\Delta t)$. This is approximately $y(t_j)+y'(t_j)\Delta t$. The derivative is usually given.

	Given $\frac{\mathrm{d}y}{\mathrm{d}t} =f(y,t)$, $y(t_0)=y_0$,
	\begin{itemize}
		\item Pick a grid size $\Delta t$.
		\item Set $Y_0=y_0$.
		\item Set $Y_1=Y_0+f(Y_0,t_0)\Delta t$.
		\item Set $Y_2=Y_1+f(Y_1,t_1)\Delta t$.
		\item Set $Y_{n+1}=Y_n+f(Y_n,t_n)\Delta t$
	\end{itemize}
	In this case, $t_{n+1}=t_n+\Delta t$.
\end{prev}
\begin{theorem}[Taylor's theorem]\label{thm:1}
	For some $f : \mathbb{R} \to \mathbb{R}$, then
	\[
		f(t)=T_n(t)+R_{n+1}(t)
	.\]
	If $f(t)$ is $n+1$ times continuously differentiable on $\left( a-\varepsilon  ,a+\varepsilon \right) $, then there exists a constant $C_{n+1}$ such that $|f(t)-T_n(t)|=\left| R_{n+1}(t) \right| \leq C_{n+1}\left| t-a \right| ^{n+1}$ for all $a-\varepsilon <t<a+\varepsilon $.

	\[C_{n+1}\coloneqq \max \left\{ \left| f^{(n+1)}[a-\varepsilon  ,a+\varepsilon ]  \right|\right\} .\]
\end{theorem}
\begin{proof}
	By the FTC,
	\[
		f(t)-f(a)=\int_{a}^{t} f'(s_1) \,\mathrm{d} s_1 = \int_{a}^{t} f'(s_1)-f'(a)+f'(a) \,\mathrm{d} s_1
	.\]
	We thus have
	\[
		f(t)=f(a)+\int_{a}^{t} f'(s_1)-f'(a)+f'(a) \,\mathrm{d} s_1=f(a)+\int_{a}^{t} f'(s_1)-f'(a) \,\mathrm{d} s_1+(t-a)f'(a)
	.\]
	Rearranging, we have
	\[
		f(t)=f(a)+f'(a)(t-a)+\int_{a}^{t} f'(s_1)-f'(a) \,\mathrm{d} s_1
	.\]
	Define
	\[		R_{2}(t)\coloneqq \int_{a}^{t} f'(s_1)-f'(a) \,\mathrm{d}s_1 
	.\]
	From the FTC, we have
	\[
		\int_{a}^{t} f'(s_1)-f'(a) \,\mathrm{d} s_1=\int_{a}^{t}\int_{a}^{s_1} f''(s_2) \,\mathrm{d} s_2 \mathrm{d} s_1
	.\]
	We also know that for any real-valued function $\phi $,
	\[
		\left| \int_{t_1}^{t_2} \phi (t) \,\mathrm{d} t \right| \leq \int_{t_1}^{t_2} \left| \phi (t) \right|  \,\mathrm{d} t
	.\]
	Thus,
	\[
		R_2(t)\leq \left| R_{2}(t) \right| =\left| \int_{a}^{t}\int_{a}^{s_1} f''(s_2) \,\mathrm{d} s_2 \mathrm{d} s_1 \right| \leq \int_{a}^{t} \left| \int_{a}^{s_1} f''(s_2) \,\mathrm{d} s_2 \right|  \,\mathrm{d} s_1
	.\]
	Now let
	\[
		M_2 \coloneqq \max_{q\in[a,t]}\left|f''[a,t]\right|
	.\]
	It follows naturally that
	\[
		\int_{a}^{t} \left| \int_{a}^{s_1} f''(s_2) \,\mathrm{d} s_2 \right|  \,\mathrm{d} s_1 \leq \int_{a}^{t}\int_{a}^{s_1} M_2 \,\mathrm{d} s_2 \mathrm{d} s_1=M_2\int_{a}^{t}\int_{a}^{s_1}  \,\mathrm{d} s_2 \mathrm{d} s_1 = M_2\int_{a}^{t} (s_1-a) \,\mathrm{d} s_1=\frac{M_2}{2} (t-a)^2
	.\]
	Therefore,
	\[
		R_2(t)\leq M_2 \frac{(t-a)^2}{2!}
	\]
	and thus,
	\[
		f(t)\leq T_1(t)+ M_2 \frac{(t-a)^2}{2!}
	\]
	in an $\varepsilon$-neighborhood of $a$. This satisfies \ref{thm:1} for $\eta =1$. Now we take an arbitrary $\eta =\xi-1 $. (this is equivalent to taking $\eta =\xi $ in the method of induction. I chose $\xi -1$ for simplicity of notation.)
	
	Suppose
	\[
		f(t)=T_{\xi -1}+R_\xi (t)=\sum_{n=1}^{\xi -1} \left( \frac{f^{(n)}(a)}{n!}(t-a)^n\right) + \int_{a}^{t} \int_{a}^{s_1} \cdots \int_{a}^{s_{\xi -1} } f^{(\xi )}(s_\xi ) \,\mathrm{d} s_\xi \cdots \,\mathrm{d} s_2 \,\mathrm{d} s_1
	.\]
	By the method of induction, it suffices to show
	\[
		\int_{a}^{t} \int_{a}^{s_1} \cdots \int_{a}^{s_{\xi -1} } f^{(\xi )}(s_\xi ) \,\mathrm{d} s_\xi \cdots \,\mathrm{d} s_2 \,\mathrm{d} s_1 = \frac{f^{(\xi )}(a)}{\xi !}(t-a)^\xi +\int_{a}^{t} \int_{a}^{s_1} \cdots \int_{a}^{s_{\xi} } f^{(\xi+1 )}(s_{\xi +1} ) \,\mathrm{d} s_{\xi +1} \cdots \,\mathrm{d} s_2 \,\mathrm{d} s_1
	,\]
	and
	\[
		\int_{a}^{t} \int_{a}^{s_1} \cdots \int_{a}^{s_{\xi} } f^{(\xi+1 )}(s_{\xi +1} ) \,\mathrm{d} s_{\xi +1} \cdots \,\mathrm{d} s_2 \,\mathrm{d} s_1\leq \max_{s_{\xi +1}\in[a,t]}\left| f^{(\xi +1)}[a,t] \right| 
	.\]
	For simplicity, we will denote $\tau +1$ repeated integrals on the interval $[a,t]$ of some function $\phi (t)$ as
	\[
		\left(J_{[a,t]}^{\tau +1}\phi\right)(t)\coloneqq \int_{a}^{t} \int_{a}^{s_1}\cdots \int_{a}^{s_\tau  } \phi (t) \,\mathrm{d} s_\tau \cdots \mathrm{d} s_1\mathrm{d} t
	.\]
	We know that
	\[
		\left( J_{[a,t]}^{\xi }f^{(\xi )} \right) (t)=\left( J_{[a,t]}^\xi g \right) (t)	
	\]
	for $g(t)=f^{(\xi )}(t)-f^{(\xi )}(a)+f^{(\xi )}(a)$. From this, by the FTC using steps seen previously, it follows that
	\[
		R_\xi (t)=\left( J_{[a,t]}^{\xi }g \right) (t)=\left( J_{[a,t]}^{\xi +1}f^{(\xi +1)} \right) (t)+\left( J_{[a,t]}^{\xi } \right)(t) \cdot f^{(\xi )}(a)
	.\]
	We leave it as an exercise to the reader to show
	\[
		\left( J^{\xi }_{[a,t]} \right)(t) =\frac{(t-a)^{\xi }}{\xi !}
	.\]
	(hint: u-sub and induct on it!)

	We  then have
	\[
	R_\xi (t)=\frac{f^{(\xi )}(t)}{\xi !}(t-a)^{\xi }+\left( J_{[a,t]}^{\xi +1} f^{(\xi +1)}\right) (t)
	.\]
	We can rewrite this as
	\[
		\left( J_{[a,t]}^{\xi +1}f^{(\xi +1)} \right) (t)=\left( J^{\xi}_{[a,t]}\int_{a}^{\xi } f^{(\xi +1)}(s_{\xi +1}) \,\mathrm{d} s_{\xi +1} \right) (t)
	.\]
	The first part of the proof has been clearly satisfied. Recall that for some real-valued $\phi : \mathbb{R} \to \mathbb{R}$,
	\[
		\int_{t_1}^{t_2} \phi (t) \,\mathrm{d} t\leq \left|\int_{t_1}^{t_2} \phi (t) \,\mathrm{d} t\right|
	,\]
	and
	\[
		\left|\int_{t_1}^{t_2} \phi (t) \,\mathrm{d} t\right| \leq \int_{t_1}^{t_2} \left| \phi (t) \right|  \,\mathrm{d} t
	.\]
	Applying this first fact, followed by repeated applications of this second fact give us
	\begin{multline*}
		\left( J^{\xi}_{[a,t]}\int_{a}^{\xi } f^{(\xi +1)}(s_{\xi +1}) \,\mathrm{d} s_{\xi +1} \right) (t)\leq \left|\left( J^{\xi}_{[a,t]}\int_{a}^{\xi } f^{(\xi +1)}(s_{\xi +1}) \,\mathrm{d} s_{\xi +1} \right) (t)\right|\\
\leq \left( J^{\xi}_{[a,t]}\int_{a}^{\xi } \left|f^{(\xi +1)}(s_{\xi +1})\right| \,\mathrm{d} s_{\xi +1} \right) (t)
	\end{multline*}	
	Define
	\[
		M_{\xi +1}\coloneqq \max_{s_\xi \in[a,t]}\left| f^{(\xi +1)}[a,t] \right| 
	.\]
	It follows naturally that
	\[
		\left( J^{\xi}_{[a,t]}\int_{a}^{\xi } \left|f^{(\xi +1)}(s_{\xi +1})\right| \,\mathrm{d} s_{\xi +1} \right) (t) \leq \left( J_{[a,t]}^{\xi }\int_{a}^{\xi } M_{\xi +1} \,\mathrm{d} s_{\xi +1} \right) (t)=M\left( J_{[a,t]}^{\xi +1} \right) (t)=\frac{M_{\xi +1}}{(\xi +1)!}(t-a)^{\xi +1}
	.\]
	Therefore,
	\[
		R_{\xi +1}(t)\leq \frac{M_{\xi +1}}{(\xi +1)!}(t-a)^{\xi +1}
	.\]
	Thus, we are done.
\end{proof}
