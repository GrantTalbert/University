\begin{theorem}[Rouche]\label{thm:?}
	Let $f,g$ be holomorphic on $\overline{D} $, where $D = B_r(z_0)$. Suppose that $\left| f-g \right| <\left| f \right| $ on $\partial D$. Then $f$ and $g$ have the same number of zeros, counted with mutliplicity, in $D$.
\end{theorem}
